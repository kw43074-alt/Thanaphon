// Google Apps Script สำหรับเชื่อมต่อ Google Sheets
// แถวที่ 1 ต้องมีหัวตาราง: id | subject | title | deadline | status

function jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
}

function findRowById(sheet, id) {
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return -1;
  }

  // อ่านเฉพาะคอลัมน์ ID ตั้งแต่แถวที่ 2
  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues();

  for (let i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(id)) {
      return i + 2;
    }
  }

  return -1;
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({
        status: "error",
        message: "ไม่พบข้อมูลที่ส่งมา"
      });
    }

    const sheet = getSheet();
    const data = JSON.parse(e.postData.contents);
    const action = data.action;

    if (action === "add") {
      if (!data.subject || !data.title || !data.deadline) {
        return jsonResponse({
          status: "error",
          message: "ข้อมูลไม่ครบ"
        });
      }

      const id = Utilities.getUuid();

      sheet.appendRow([
        id,
        String(data.subject).trim(),
        String(data.title).trim(),
        String(data.deadline),
        data.status || "Pending"
      ]);

      return jsonResponse({
        status: "success",
        message: "เพิ่มข้อมูลสำเร็จ",
        id: id
      });
    }

    if (action === "updateTask") {
      const rowNumber = findRowById(sheet, data.id);

      if (rowNumber === -1) {
        return jsonResponse({
          status: "error",
          message: "ไม่พบ ID ที่ต้องการแก้ไข"
        });
      }

      if (!data.subject || !data.title || !data.deadline) {
        return jsonResponse({
          status: "error",
          message: "ข้อมูลสำหรับแก้ไขไม่ครบ"
        });
      }

      // แก้ไข subject, title, deadline และคง/อัปเดต status
      sheet.getRange(rowNumber, 2, 1, 4).setValues([[
        String(data.subject).trim(),
        String(data.title).trim(),
        String(data.deadline),
        data.status || "Pending"
      ]]);

      return jsonResponse({
        status: "success",
        message: "แก้ไขข้อมูลสำเร็จ"
      });
    }

    if (action === "updateStatus" || action === "update") {
      const rowNumber = findRowById(sheet, data.id);

      if (rowNumber === -1) {
        return jsonResponse({
          status: "error",
          message: "ไม่พบ ID ที่ต้องการเปลี่ยนสถานะ"
        });
      }

      sheet.getRange(rowNumber, 5).setValue(data.status);

      return jsonResponse({
        status: "success",
        message: "เปลี่ยนสถานะสำเร็จ"
      });
    }

    if (action === "delete") {
      const rowNumber = findRowById(sheet, data.id);

      if (rowNumber === -1) {
        return jsonResponse({
          status: "error",
          message: "ไม่พบ ID ที่ต้องการลบ"
        });
      }

      sheet.deleteRow(rowNumber);

      return jsonResponse({
        status: "success",
        message: "ลบข้อมูลสำเร็จ"
      });
    }

    return jsonResponse({
      status: "error",
      message: "ไม่รู้จักคำสั่ง: " + action
    });

  } catch (error) {
    return jsonResponse({
      status: "error",
      message: error.message
    });
  }
}

function doGet() {
  try {
    const sheet = getSheet();
    const values = sheet.getDataRange().getDisplayValues();

    if (values.length === 0) {
      return jsonResponse([]);
    }

    const headers = values.shift();

    const result = values
      .filter(row => row.some(cell => cell !== ""))
      .map(row => {
        const item = {};

        headers.forEach((header, index) => {
          item[String(header).trim()] = row[index];
        });

        return item;
      });

    return jsonResponse(result);

  } catch (error) {
    return jsonResponse({
      status: "error",
      message: error.message
    });
  }
}
